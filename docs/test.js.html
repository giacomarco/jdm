    /**
 * Classe che rappresenta una persona.
 */
class Persona {
    /**
     * Crea un'istanza di una persona.
     * @param {string} nome - Il nome della persona.
     * @param {number} eta - L'età della persona.
     */
    constructor(nome, eta) {
        this.nome = nome;
        this.eta = eta;
    }

    /**
     * Restituisce una descrizione della persona.
     * @returns {string} Una stringa che descrive la persona.
     */
    descrizione() {
        return `${this.nome} ha ${this.eta} anni.`;
    }
}
